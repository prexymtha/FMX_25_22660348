# make sure data is ordered
beta_data <- 
  analysis_data %>% 
  arrange(date)

# rolling regression: y on x with width = 36
# ABC vs Benchmark
roll_abc <- 
  roll::roll_lm(
    x     = as.matrix(beta_data$Benchmark_Returns),
    y     = beta_data$ABC_Returns,
    width = 36
  )

# Peer average vs Benchmark
roll_peer <- 
  roll::roll_lm(
    x     = as.matrix(beta_data$Benchmark_Returns),
    y     = beta_data$Peers_Avg_Returns,
    width = 36
  )

# extract slope (beta) from coefficients[, 2]
rolling_betas <- 
  tibble(
    date        = beta_data$date,
    ABC_Fund    = as.numeric(roll_abc$coefficients[, 2]),
    Peer_Average = as.numeric(roll_peer$coefficients[, 2])
  ) %>% 
  pivot_longer(
    cols      = c(ABC_Fund, Peer_Average),
    names_to  = "Series",
    values_to = "Beta"
  ) %>% 
  filter(!is.na(Beta))   # drop early windows with no full 36 months

roll_df <-
  analysis_data %>%
  arrange(date) %>%
  transmute(
    date,
    Benchmark    = Benchmark_Returns,
    ABC_Fund     = ABC_Returns,
    Peer_Average = Peers_Avg_Returns
  ) %>%
  pivot_longer(
    cols      = -date,
    names_to  = "Series",
    values_to = "ret"
  ) %>%
  group_by(Series) %>%
  # make sure NAs don’t break the product
  mutate(
    ret      = coalesce(ret, 0),
    RollRets = RcppRoll::roll_prod(1 + ret, 36, fill = NA, align = "right")^(12 / 36) - 1
  ) %>%
  # keep only dates where at least one series has a rolling return
  group_by(date) %>%
  filter(any(!is.na(RollRets))) %>%
  ungroup()

g_roll <-
  roll_df %>%
  ggplot(aes(x = date, y = RollRets, colour = Series)) +
  geom_line(alpha = 0.7, linewidth = 1.1) +
  labs(
    title = "Rolling 3-year annualised returns",
    x     = "",

  analysis_data %>%

  transmute(
    date,

    Peer_Average = Peers_Avg_Returns

  pivot_longer(
    cols      = -date,
    names_to  = "Series",
    values_to = "ret"
  ) %>%
  # already monthly data, so no need for YM collapsing
  arrange(date) %>%
  group_by(Series) %>%
  ungroup()

#========================================================
# 2. Manual calculation of 6m, 12m, 3y, 5y annualised returns
#========================================================

dfplot <-
  bind_rows(
    # 6 months (not annualised in Nico's example)
    idx %>%
      filter(date >= fmxdat::safe_month_min(last(date), N = 6)) %>%
      group_by(Series) %>%
      summarise(mu = prod(1 + ret, na.rm = TRUE) - 1) %>%
      mutate(Freq = "A"),
    # 12 months (annualised over 1 year => exponent 12/12)
    idx %>%
      filter(date >= fmxdat::safe_month_min(last(date), N = 12)) %>%
      group_by(Series) %>%
      summarise(mu = prod(1 + ret, na.rm = TRUE)^(12 / 12) - 1) %>%
      mutate(Freq = "B"),
    # 3 years (36 months, annualised)
    idx %>%
      filter(date >= fmxdat::safe_month_min(last(date), N = 36)) %>%
      group_by(Series) %>%
      summarise(mu = prod(1 + ret, na.rm = TRUE)^(12 / 36) - 1) %>%
      mutate(Freq = "C"),
    # 5 years (60 months, annualised)
    idx %>%
      filter(date >= fmxdat::safe_month_min(last(date), N = 60)) %>%
      group_by(Series) %>%
      summarise(mu = prod(1 + ret, na.rm = TRUE)^(12 / 60) - 1) %>%
      mutate(Freq = "D")
  )
